cmake_minimum_required(VERSION 3.20)
project(cadkernel VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()

# address = ASan + UBSan, thread = TSan. They can't be combined.
set(CAD_SANITIZE "" CACHE STRING "Sanitizer: '', address, thread")
option(CAD_WERROR "Treat warnings as errors" OFF)
option(CAD_BUILD_EXAMPLES "Build demo and benchmark programs" ON)
option(CAD_BUILD_BUG_DEMO "Build the deliberately broken ASan demo" OFF)

add_library(cad_warnings INTERFACE)
if(MSVC)
  target_compile_options(cad_warnings INTERFACE /W4 /permissive- $<$<BOOL:${CAD_WERROR}>:/WX>)
else()
  target_compile_options(cad_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow
    $<$<BOOL:${CAD_WERROR}>:-Werror>)
endif()

if(CAD_SANITIZE AND NOT MSVC)
  if(CAD_SANITIZE STREQUAL "address")
    set(_san -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  elseif(CAD_SANITIZE STREQUAL "thread")
    set(_san -fsanitize=thread)
  else()
    message(FATAL_ERROR "CAD_SANITIZE must be empty, 'address' or 'thread'")
  endif()
  add_compile_options(${_san} -fno-omit-frame-pointer -g)
  add_link_options(${_san})
endif()

file(GLOB_RECURSE GEOM_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(geom ${GEOM_SOURCES})
target_include_directories(geom PUBLIC include)
target_link_libraries(geom PRIVATE cad_warnings)
find_package(Threads REQUIRED)
target_link_libraries(geom PUBLIC Threads::Threads)

include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)

  file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
  add_executable(geom_tests ${TEST_SOURCES})
  target_link_libraries(geom_tests PRIVATE geom cad_warnings GTest::gtest_main)
  if(CAD_SANITIZE STREQUAL "thread")
    # TSan aborts on kernels with high mmap ASLR entropy ("unexpected memory
    # mapping"), e.g. recent WSL2. Run it with ASLR disabled instead of
    # discovering tests, since discovery runs the binary at build time.
    find_program(SETARCH setarch)
    if(SETARCH)
      add_test(NAME geom_tests COMMAND ${SETARCH} -R $<TARGET_FILE:geom_tests>)
    else()
      add_test(NAME geom_tests COMMAND geom_tests)
    endif()
  else()
    include(GoogleTest)
    gtest_discover_tests(geom_tests)
  endif()
endif()

if(CAD_BUILD_EXAMPLES)
  foreach(ex unfold_demo bench)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/examples/${ex}.cpp)
      add_executable(${ex} examples/${ex}.cpp)
      target_link_libraries(${ex} PRIVATE geom cad_warnings)
    endif()
  endforeach()
endif()

if(CAD_BUILD_BUG_DEMO)
  add_executable(bug_demo examples/bug_demo/use_after_free.cpp)
  target_link_libraries(bug_demo PRIVATE geom)
endif()
